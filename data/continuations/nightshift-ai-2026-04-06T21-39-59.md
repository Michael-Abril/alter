NightShift AI differentiates from competitors like Poke by being a **continuation engine** rather than a reactive assistant. While Poke waits for signals and nudges users, NightShift becomes the user—analyzing half-finished work, understanding context across connected apps, and completing tasks autonomously overnight.

```python
# Core context understanding system
class ContextEngine:
    def __init__(self):
        self.pinecone_client = PineconeClient()
        self.embedding_model = OpenAIEmbeddings()
    
    def analyze_unfinished_work(self, user_id: str):
        """Detect incomplete emails and documents"""
        gmail_context = self.get_email_threads(user_id)
        docs_context = self.get_document_drafts(user_id)
        
        return {
            'pending_emails': self.find_pending_responses(gmail_context),
            'incomplete_docs': self.find_incomplete_documents(docs_context)
        }
    
    def get_continuation_context(self, task_id: str):
        """Retrieve relevant context for task completion"""
        similar_contexts = self.pinecone_client.query(
            vector=self.embed_task(task_id),
            top_k=10
        )
        return self.synthesize_context(similar_contexts)
```

### 2. Email Continuation Service
```python
class EmailContinuationService:
    def __init__(self):
        self.claude_client = ClaudeClient()
        self.style_analyzer = WritingStyleAnalyzer()
    
    async def draft_email_response(self, thread_id: str, user_id: str):
        # Get conversation context
        thread = await self.get_email_thread(thread_id)
        user_style = await self.style_analyzer.get_user_style(user_id)
        
        # Generate response with Claude
        response = await self.claude_client.complete(
            prompt=self.build_email_prompt(thread, user_style),
            model="claude-3-5-sonnet-20241022"
        )
        
        return EmailDraft(
            thread_id=thread_id,
            content=response.content,
            confidence_score=self.calculate_confidence(response)
        )
```

### 3. Document Continuation Service
```python
class DocumentContinuationService:
    def __init__(self):
        self.gdocs_client = GoogleDocsClient()
        self.claude_client = ClaudeClient()
    
    async def continue_document(self, doc_id: str, user_id: str):
        # Analyze document structure and content
        doc_content = await self.gdocs_client.get_document(doc_id)
        writing_style = await self.analyze_document_style(doc_content)
        
        # Generate continuation
        continuation = await self.claude_client.complete(
            prompt=self.build_document_prompt(doc_content, writing_style),
            model="claude-3-5-sonnet-20241022"
        )
        
        return DocumentContinuation(
            doc_id=doc_id,
            content=continuation.content,
            insertion_point=self.find_insertion_point(doc_content)
        )
```

### 4. OpenClaw Integration Layer
```python
class OpenClawOrchestrator:
    def __init__(self):
        self.openclaw = OpenClawClient()
    
    async def execute_email_action(self, action: EmailAction):
        """Execute email actions via OpenClaw"""
        if action.type == "send_draft":
            return await self.openclaw.gmail.send_email(
                to=action.recipient,
                subject=action.subject,
                body=action.content,
                thread_id=action.thread_id
            )
    
    async def execute_document_action(self, action: DocumentAction):
        """Execute document actions via OpenClaw"""
        if action.type == "append_content":
            return await self.openclaw.gdocs.append_text(
                document_id=action.doc_id,
                text=action.content,
                position=action.insertion_point
            )
```

## Database Schema

### Core Tables
```sql
-- Users and authentication
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    clerk_user_id VARCHAR UNIQUE NOT NULL,
    email VARCHAR NOT NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    preferences JSONB DEFAULT '{}'::jsonb
);

-- Connected accounts (Gmail, Google Docs)
CREATE TABLE connected_accounts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    provider VARCHAR NOT NULL, -- 'gmail', 'gdocs'
    account_id VARCHAR NOT NULL,
    access_token_encrypted TEXT,
    refresh_token_encrypted TEXT,
    expires_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Task queue for overnight work
CREATE TABLE pending_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    task_type VARCHAR NOT NULL, -- 'email_response', 'document_continuation'
    source_id VARCHAR NOT NULL, -- thread_id or doc_id
    context JSONB,
    priority INTEGER DEFAULT 0,
    status VARCHAR DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT NOW(),
    scheduled_for TIMESTAMP
);

-- Completed work for morning review
CREATE TABLE completed_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    original_task_id UUID REFERENCES pending_tasks(id),
    task_type VARCHAR NOT NULL,
    result JSONB,
    confidence_score DECIMAL(3,2),
    user_approved BOOLEAN,
    feedback TEXT,
    completed_at TIMESTAMP DEFAULT NOW()
);
```
